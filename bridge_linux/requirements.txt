# Two backends, picked automatically at startup — see "Hand skeleton backends"

# Monado / WiVRn: the native OpenXR backend. Required there, because an
# OpenVR-on-OpenXR layer (xrizer, OpenComposite) structurally cannot feed a
# non-rendering background app — this one uses XR_MND_headless +
# XR_EXT_hand_tracking instead. Pulls numpy/glfw/pyopengl transitively; the
# graphics ones go unused in a headless session.
pyopenxr

# SteamVR: the OpenVR skeletal-input backend.
openvr
